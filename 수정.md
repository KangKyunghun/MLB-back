private void saveStandingsAndTeams(League league, Season season, int year) {
    try {
        var res = webClient.get()
            .uri("/competitions/{id}/standings?season={year}", league.getId(), year)
            .retrieve()
            .bodyToMono(FootballApiDto.StandingsResponse.class)
            .block();

        if (res == null || res.getStandings() == null) return;

        res.getStandings().stream()
            .filter(s -> "TOTAL".equals(s.getType()))
            .findFirst()
            .ifPresent(table -> {

                for (var dto : table.getTable()) {

                    // 🔥 팀 없으면 생성
                    Team team = teamRepository.findById(dto.getTeam().getId())
                        .orElseGet(() -> teamRepository.save(
                            Team.builder()
                                .id(dto.getTeam().getId())
                                .name(dto.getTeam().getName())
                                .league(league)
                                .build()
                        ));

                    // 🔥 순위 저장
                    standingRepository.findBySeasonIdAndTeamId(season.getId(), team.getId())
                        .orElseGet(() -> standingRepository.save(
                            Standing.builder()
                                .league(league)
                                .season(season)
                                .team(team)
                                .rank(dto.getPosition())
                                .points(dto.getPoints())
                                .played(dto.getPlayedGames())
                                .build()
                        ));
                }
            });

        log.info("standings 완료: {} {}", league.getId(), year);

    } catch (Exception e) {
        log.warn("standings 실패: {} {}", league.getId(), year);
    }
}

private void saveMatches(League league, Season season, int year) {

    LocalDate start = LocalDate.of(year, 8, 1);
    LocalDate end = LocalDate.of(year + 1, 5, 31);

    while (start.isBefore(end)) {

        LocalDate chunkEnd = start.plusMonths(2); // 🔥 2개월 단위

        fetchAndSaveMatches(league, season, start, chunkEnd);

        start = chunkEnd;
    }
}
